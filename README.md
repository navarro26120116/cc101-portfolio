Sample Pseudocode: Decimal to Binary

START
    INPUT number
    IF number = 0 THEN
        DISPLAY "Binary: 0"
    ELSE
        SET binary = ""
        WHILE number > 0 DO
            SET remainder = number MOD 2
            SET binary = remainder + binary
            SET number = number DIV 2
        ENDWHILE
        DISPLAY "Binary: ", binary
    ENDIF
END
